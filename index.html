create extension if not exists pgcrypto; -- أنواع البيانات create type public.app_role as enum ( 'member', 'treasurer', 'reviewer', 'admin' ); create type public.member_status as enum ( 'pending', 'active', 'suspended', 'closed' ); create type public.due_status as enum ( 'unpaid', 'partially_paid', 'paid', 'waived' ); create type public.payment_status as enum ( 'pending', 'confirmed', 'rejected', 'refunded' ); create type public.payment_method as enum ( 'cash', 'bank_transfer', 'electronic', 'other' ); create type public.help_request_status as enum ( 'submitted', 'under_review', 'approved', 'rejected', 'paid', 'cancelled' ); create type public.journal_status as enum ( 'draft', 'posted', 'void' ); -- ملف المستخدم؛ كلمة المرور لا تُخزن هنا create table public.profiles ( id uuid primary key references auth.users(id) on delete cascade, full_name text not null, phone text, email text, active boolean not null default true, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- يمكن منح المستخدم أكثر من دور create table public.user_roles ( user_id uuid not null references public.profiles(id) on delete cascade, role public.app_role not null, granted_by uuid references public.profiles(id) on delete set null, created_at timestamptz not null default now(), primary key (user_id, role) ); -- بيانات العضوية create table public.members ( id uuid primary key default gen_random_uuid(), profile_id uuid unique references public.profiles(id) on delete set null, membership_number text not null unique, status public.member_status not null default 'pending', join_date date, address text, notes text, created_by uuid references public.profiles(id) on delete set null, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- خطط الاشتراك create table public.contribution_plans ( id uuid primary key default gen_random_uuid(), name text not null, amount numeric(12,2) not null check (amount > 0), frequency text not null check (frequency in ('monthly', 'quarterly', 'yearly', 'one_time')), active boolean not null default true, effective_from date not null default current_date, effective_to date, created_at timestamptz not null default now(), check (effective_to is null or effective_to >= effective_from) ); -- المبالغ المستحقة على الأعضاء create table public.dues ( id uuid primary key default gen_random_uuid(), member_id uuid not null references public.members(id) on delete restrict, plan_id uuid references public.contribution_plans(id) on delete set null, period_start date not null, period_end date not null, amount_due numeric(12,2) not null check (amount_due > 0), due_date date, status public.due_status not null default 'unpaid', created_at timestamptz not null default now(), check (period_end >= period_start), unique (member_id, period_start, period_end) ); -- دفعات الاشتراك أو التبرعات create table public.payments ( id uuid primary key default gen_random_uuid(), member_id uuid not null references public.members(id) on delete restrict, amount numeric(12,2) not null check (amount > 0), method public.payment_method not null default 'cash', status public.payment_status not null default 'pending', received_at timestamptz, received_by uuid references public.profiles(id) on delete set null, reference_number text, note text, idempotency_key text unique, created_at timestamptz not null default now() ); -- ربط الدفعة بالاشتراكات؛ يسمح بدفعة واحدة لتغطية أكثر من فترة create table public.payment_allocations ( id uuid primary key default gen_random_uuid(), payment_id uuid not null references public.payments(id) on delete restrict, due_id uuid not null references public.dues(id) on delete restrict, amount numeric(12,2) not null check (amount > 0), created_at timestamptz not null default now(), unique (payment_id, due_id) ); -- طلبات المساعدة create table public.help_requests ( id uuid primary key default gen_random_uuid(), member_id uuid not null references public.members(id) on delete restrict, request_type text not null, requested_amount numeric(12,2) check (requested_amount is null or requested_amount > 0), reason text not null, status public.help_request_status not null default 'submitted', submitted_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- قرارات وملاحظات المراجعة؛ يحتفظ بسجل المراجعات المتتابعة create table public.request_reviews ( id uuid primary key default gen_random_uuid(), request_id uuid not null references public.help_requests(id) on delete cascade, reviewer_id uuid references public.profiles(id) on delete set null, decision text not null check (decision in ('under_review', 'approved', 'rejected')), note text, decided_at timestamptz not null default now() ); -- المرفقات: الملف نفسه يُرفع إلى Storage create table public.attachments ( id uuid primary key default gen_random_uuid(), owner_user_id uuid references public.profiles(id) on delete set null, request_id uuid references public.help_requests(id) on delete cascade, payment_id uuid references public.payments(id) on delete cascade, storage_path text not null, file_name text, content_type text, uploaded_at timestamptz not null default now(), check (request_id is not null or payment_id is not null) ); -- الحسابات المالية create table public.ledger_accounts ( id uuid primary key default gen_random_uuid(), code text not null unique, name text not null, account_type text not null check ( account_type in ('asset', 'liability', 'income', 'expense', 'equity') ), active boolean not null default true ); -- قيد محاسبي رئيسي، مثل تحصيل أو صرف مساعدة create table public.journal_entries ( id uuid primary key default gen_random_uuid(), entry_date date not null default current_date, description text not null, status public.journal_status not null default 'draft', source_type text, source_id uuid, created_by uuid references public.profiles(id) on delete set null, created_at timestamptz not null default now(), posted_at timestamptz, unique (source_type, source_id) ); -- تفاصيل القيد المحاسبي create table public.journal_lines ( id uuid primary key default gen_random_uuid(), journal_entry_id uuid not null references public.journal_entries(id) on delete cascade, account_id uuid not null references public.ledger_accounts(id) on delete restrict, debit numeric(12,2) not null default 0 check (debit >= 0), credit numeric(12,2) not null default 0 check (credit >= 0), description text, check ( (debit > 0 and credit = 0) or (credit > 0 and debit = 0) ) ); -- صرف المساعدة create table public.disbursements ( id uuid primary key default gen_random_uuid(), request_id uuid not null references public.help_requests(id) on delete restrict, amount numeric(12,2) not null check (amount > 0), method public.payment_method not null default 'bank_transfer', paid_at timestamptz, paid_by uuid references public.profiles(id) on delete set null, reference_number text, journal_entry_id uuid unique references public.journal_entries(id) on delete set null, note text, created_at timestamptz not null default now() ); -- إشعارات التطبيق create table public.notifications ( id uuid primary key default gen_random_uuid(), user_id uuid not null references public.profiles(id) on delete cascade, title text not null, message text not null, read_at timestamptz, created_at timestamptz not null default now() ); -- سجل تغييرات الإدارة والعمليات الحساسة create table public.audit_logs ( id bigint generated always as identity primary key, actor_user_id uuid references public.profiles(id) on delete set null, action text not null, entity_type text not null, entity_id uuid, details jsonb not null default '{}'::jsonb, created_at timestamptz not null default now() ); -- فهارس لتحسين البحث والتقارير create index idx_members_status on public.members(status); create index idx_dues_member_period on public.dues(member_id, period_start); create index idx_dues_status_date on public.dues(status, due_date); create index idx_payments_member_date on public.payments(member_id, created_at desc); create index idx_payments_status on public.payments(status); create index idx_help_requests_member_date on public.help_requests(member_id, submitted_at desc); create index idx_help_requests_status_date on public.help_requests(status, submitted_at desc); create index idx_notifications_user_unread on public.notifications(user_id, created_at desc) where read_at is null; create index idx_journal_entries_date on public.journal_entries(entry_date); create index idx_audit_logs_entity on public.audit_logs(entity_type, entity_id);
